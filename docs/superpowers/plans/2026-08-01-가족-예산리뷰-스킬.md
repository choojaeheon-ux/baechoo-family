```js
export async function resolve(spec, ctx, next) {
  if (spec.startsWith(".") && !HAS_EXT.test(spec)) {

      // 확장자가 붙은 형태가 없으면 원래 스펙으로 넘어간다
```

```js
import fs from "node:fs";
import path from "node:path";

const PAGE = 1000;

export function parseEnv(text) {
  const out = {};
  for (const line of text.split(/\r?\n/)) {
    if (!line || line.startsWith("#")) continue;
    const i = line.indexOf("=");
    if (i < 0) continue;
    out[line.slice(0, i).trim()] = line.slice(i + 1).trim(); // 값 안의 '='는 그대로
  }
  return { url: out.NEXT_PUBLIC_SUPABASE_URL, key: out.NEXT_PUBLIC_SUPABASE_ANON_KEY };
}

export function readEnv(appDir) {
  const p = path.join(appDir, ".env.local");
  if (!fs.existsSync(p)) throw new Error(`.env.local 없음: ${p}`);
  const env = parseEnv(fs.readFileSync(p, "utf8"));
  if (!env.url || !env.key) throw new Error(`.env.local에 SUPABASE URL/ANON_KEY 없음: ${p}`);
  return env;
}

// PostgREST 응답 상한에 걸려 조용히 잘리지 않도록 페이지 단위로 끝까지 읽는다.
export async function fetchAll(env, table, query = "") {
  const rows = [];
  for (let offset = 0; ; offset += PAGE) {
    const url = `${env.url}/rest/v1/${table}?select=*${query}&limit=${PAGE}&offset=${offset}`;
    const res = await fetch(url, {
      headers: { apikey: env.key, Authorization: `Bearer ${env.key}` },
    });
    if (!res.ok) throw new Error(`${table} 조회 실패 ${res.status}: ${(await res.text()).slice(0, 200)}`);
    const page = await res.json();
    rows.push(...page);
    if (page.length < PAGE) return rows;
  }
}
```

```js
export function unknownNames(names, categories) {

export function splitAnnual(rows, annualNames) {
  const set = new Set(annualNames);
  const monthly = [], annual = [];
  for (const r of rows) (set.has(r.category.name) ? annual : monthly).push(r);
  return { monthly, annual };

// 연초~당월 누적 지출. 문자열 비교로 충분하다(날짜가 전부 YYYY-MM-DD).
function ytd(txns, categoryId, ym) {
  const from = ym.slice(0, 4) + "-01-01";
  const to = ym + "-32"; // "YYYY-MM-32"는 그 달의 모든 날짜보다 크다
  let sum = 0;
  for (const t of txns) {
    if (t.type !== "expense" || t.categoryId !== categoryId) continue;
    if (t.date >= from && t.date <= to) sum += t.amount;
  }
  return sum;
}

export function annualRows(annualSpecs, categories, budgets, txns, ym) {
  const m = byName(categories);
  const out = [];
  for (const spec of annualSpecs) {
    const c = m.get(spec.category);

    const monthBudget = budgetForCategory(budgets, ym, c.id) ?? 0;
    const yearBudget = monthBudget * 12;

    out.push({
      name: c.name,
      group: c.groupName,
      kind: spec.kind,
      note: spec.note ?? null,
```

```js
import { monthTransactions, spendByCategory } from "../../../baechoo-family/lib/compute.ts";
import { shiftMonth } from "../../../baechoo-family/lib/format.ts";

const TOP_ITEMS = 5;

export function txLabel(t) {
  if (t.merchant && t.memo) return `${t.merchant} · ${t.memo}`;
  return t.merchant || t.memo || "(내용 없음)";
}

// 당월 포함 최근 N개월 평균. 거래가 없는 달도 분모에 넣는다(0원으로 친다).
function avgOf(txns, categoryId, ym, months) {
  let sum = 0;
  for (let i = 0; i < months; i++) {
    const m = shiftMonth(ym, -i);
    sum += spendByCategory(monthTransactions(txns, m)).get(categoryId) ?? 0;
  }
  return months > 0 ? sum / months : 0;
}

export function unbudgetedRows(monthlyRows, txns, ym, avgMonths) {
  const cur = monthTransactions(txns, ym);
  return monthlyRows
    .filter((r) => r.budget === 0 && r.spend > 0)
    .sort((a, b) => b.spend - a.spend)
    .map((r) => ({
      name: r.category.name,
      group: r.category.groupName,
      spend: r.spend,
      avgSpend: avgOf(txns, r.category.id, ym, avgMonths),
      items: cur
        .filter((t) => t.type === "expense" && t.categoryId === r.category.id)
        .sort((a, b) => b.amount - a.amount)
        .slice(0, TOP_ITEMS)
        .map((t) => ({ date: t.date, amount: t.amount, label: txLabel(t) })),
    }));
}
```

```js
  const picked = monthlyRows

    underspend: { ...DEFAULTS.underspend, ...(context.underspend ?? {}) },
  };
  const annualSpecs = ctx.annual ?? [];
  const notes = ctx.notes ?? {};

  const warnings = [
    ...unknownNames(annualSpecs.map((a) => a.category), categories)
      .map((n) => `context.annual의 과목명 "${n}"이 DB에 없음 — 무시함`),
    ...unknownNames(Object.keys(notes), categories)
      .map((n) => `context.notes의 과목명 "${n}"이 DB에 없음 — 무시함`),
  ];

  // 요약은 앱과 같은 전체값을 쓴다. 연납을 빼고 다시 계산하지 않는다
  // — 앱 화면과 숫자가 어긋나면 이 스킬의 존재 이유가 사라진다.
  const burn = budgetBurndown(budgets, categories, transactions, ym);
  const { monthly, annual } = splitAnnual(burn.rows, annualSpecs.map((a) => a.category));

  const groups = groupBurnRows(monthly).map((g) => ({
    name: g.name,
    budget: g.budget,
    spend: g.spend,
    rows: g.rows.map(flatRow),
  }));

  const categoryById = (id) => categories.find((c) => c.id === id);
  const pnl = computePnl(monthTransactions(transactions, ym), categoryById);

  return {
    ym,
    generatedAt: new Date().toISOString().slice(0, 10),
    summary: { budget: burn.budget, spend: burn.spend, pct: burn.pct, revenue: pnl.revenue },
    groups,
    annual: annualRows(annualSpecs, categories, budgets, transactions, ym),
    unbudgeted: unbudgetedRows(monthly, transactions, ym, ctx.avgMonths),
    underspent: underspentRows(monthly, ctx.underspend),
    pnl,
    waterfall: buildWaterfall(pnl),
    notes,
    warnings,
  };
}
```

`analyze.mjs` 맨 위의 import 문들은 하나로 합친다(`budgetForCategory`·`monthTransactions`·`spendByCategory`·`budgetBurndown`·`groupBurnRows`를 `compute.ts`에서 한 번에).

섹션 번호는 표시되는 섹션에만 붙인다. 카운터를 하나 두고 헤더를 낼 때마다 올린다.

```js
export const won = (n) => Math.round(n ?? 0).toLocaleString("ko-KR");

export const esc = (s) =>
  String(s ?? "")
    .replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;").replace(/'/g, "&#39;");

// 막대는 `.row` 그리드의 세 번째 칸 하나만 차지한다. 퍼센트 글자는 호출부의 `.pc`가 낸다
// — 여기서 같이 내면 4열 그리드에 5개가 들어가 금액 칸이 다음 줄로 밀린다.
// 예산 0인데 지출이 있으면 비율을 낼 수 없다. 앱이 Infinity를 가득 찬 막대로 그리므로 똑같이 맞춘다.
export function bar(pct, over) {
  const w = pct == null ? (over ? 100 : 0) : Math.min(100, Math.max(0, pct));
  return `<div class="track"><div class="fill${over ? " is-over" : ""}" style="width:${w}%"></div></div>`;
}

export function renderHtml(review, template) {
  const r = review;
  let n = 0;
  const h2 = (title) => `<h2>${++n}. ${esc(title)}</h2>`;
  const out = [];

  out.push(`<h1>배추가족 ${Number(r.ym.slice(5))}월 예산 리뷰</h1>`);
  out.push(`<div class="sub">${esc(r.ym)} · 작성 ${esc(r.generatedAt)}</div>`);

  if (r.warnings.length) {
    out.push(`<div class="warn">${r.warnings.map(esc).join("<br>")}</div>`);
  }

  const overBudget = r.summary.pct > 100;
  out.push(`<div class="kpis">

// 폭포 차트 — 인라인 SVG. 라이브러리도 canvas도 쓰지 않는다(인쇄 타이밍 문제 회피)
function waterfallSvg(segs) {
  const W = 660, H = 150, PAD = 22;
  const max = Math.max(...segs.map((s) => Math.max(s.range[0], s.range[1])), 1);
  const bw = (W - PAD * 2) / segs.length;
  const COLOR = { revenue: "#5c93a8", deduct: "#d9a441", profit: "#5b8c3e" };
  const bars = segs.map((s, i) => {
    const [lo, hi] = s.range;
    const y = H - PAD - (hi / max) * (H - PAD * 2);
    const h = Math.max(1, ((hi - lo) / max) * (H - PAD * 2));
    const x = PAD + i * bw + bw * 0.18;
    const fill = COLOR[s.kind] ?? "#7c766a";
    return `<rect x="${x.toFixed(1)}" y="${y.toFixed(1)}" width="${(bw * 0.64).toFixed(1)}" height="${h.toFixed(1)}" fill="${fill}"/>
      <text x="${(x + bw * 0.32).toFixed(1)}" y="${(y - 3).toFixed(1)}" font-size="9" text-anchor="middle" fill="#2f2a20">${won(s.value)}</text>
      <text x="${(x + bw * 0.32).toFixed(1)}" y="${H - 6}" font-size="9" text-anchor="middle" fill="#7c766a">${esc(s.label)}</text>`;
  });
  return `<svg viewBox="0 0 ${W} ${H}" width="100%" style="margin:2mm 0">${bars.join("")}</svg>`;
}
```

```js
import fs from "node:fs";
import path from "node:path";
import { execFileSync } from "node:child_process";

const CANDIDATES = [
  [process.env["ProgramFiles"], "Google/Chrome/Application/chrome.exe"],
  [process.env["ProgramFiles(x86)"], "Google/Chrome/Application/chrome.exe"],
  [process.env["LOCALAPPDATA"], "Google/Chrome/Application/chrome.exe"],
  [process.env["ProgramFiles(x86)"], "Microsoft/Edge/Application/msedge.exe"],
];

export function findChrome(explicit) {
  if (explicit) {
    if (!fs.existsSync(explicit)) throw new Error(`--chrome 경로 없음: ${explicit}`);
    return explicit;
  }
  for (const [base, rel] of CANDIDATES) {
    if (!base) continue;
    const p = path.join(base, rel);
    if (fs.existsSync(p)) return p;
  }
  // 조용히 HTML만 남기지 않는다 — PDF가 산출물이므로 실패로 알린다
  throw new Error("Chrome/Edge를 찾지 못함. --chrome <실행파일 경로>로 지정할 것");
}

export function htmlToPdf(htmlPath, pdfPath, chromePath) {
  const userData = fs.mkdtempSync(path.join(process.env.TEMP || ".", "fbr-"));
  try {
    execFileSync(chromePath, [
      "--headless=new",
      "--disable-gpu",
      "--no-pdf-header-footer",
      `--user-data-dir=${userData}`,
      `--print-to-pdf=${pdfPath}`,
      "file:///" + htmlPath.replace(/\\/g, "/"),
    ], { stdio: "pipe", timeout: 120000 });
  } finally {
    fs.rmSync(userData, { recursive: true, force: true });
  }
  if (!fs.existsSync(pdfPath)) throw new Error(`PDF가 생성되지 않음: ${pdfPath}`);
}
```
